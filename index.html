Googleスプレッドシート連携（URL設定）

1 Google Apps Script (GAS) ウェブアプリURL

先生ご自身のGoogleスプレッドシートに以下のスクリプトを設定して発行した「ウェブアプリのURL」を貼り付けると、ワンクリックでバックアップと復元が可能になります。

スプレッドシート「拡張機能 ＞ Apps Script」に貼り付けるコード： 
function doGet(e) {
  const sheet = getOrCreateSheet();
  const action = e.parameter.action || "load";
  const data = sheet.getDataRange().getValues();

  // 保存済み一覧を返す（単体＋名前付きセット）
  if (action === "list") {
    const list = [];
    for (let i = 1; i < data.length; i++) {
      if (data[i][0]) {
        const key = String(data[i][0]);
        const updatedAt = data[i][3] ? Utilities.formatDate(new Date(data[i][3]), "Asia/Tokyo", "yyyy/MM/dd HH:mm") : "";
        if (key.indexOf("SET:") === 0) {
          list.push({ key: key, kind: "set", name: key.slice(4), title: data[i][1] || "", updatedAt: updatedAt });
        } else {
          const parts = key.split("_");
          list.push({ key: key, kind: "single", grade: parts[0] || "", term: parts[1] || "", title: data[i][1] || "", updatedAt: updatedAt });
        }
      }
    }
    return jsonResponse({ status: "success", list: list });
  }

  // 名前付きセットを読み込む
  if (action === "loadSet") {
    const targetKey = "SET:" + (e.parameter.name || "");
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === targetKey) {
        return jsonResponse({ status: "success", name: e.parameter.name, data: JSON.parse(data[i][2]) });
      }
    }
    return jsonResponse({ status: "not_found" });
  }

  // 特定の学年・学期を読み込む
  const grade = e.parameter.grade || "3";
  const term = e.parameter.term || "2";
  const key = grade + "_" + term;
  
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === key) {
      return jsonResponse({
        status: "success",
        title: data[i][1],
        questions: JSON.parse(data[i][2])
      });
    }
  }
  return jsonResponse({ status: "not_found" });
}

function doPost(e) {
  try {
    const sheet = getOrCreateSheet();
    const raw = e.postData ? e.postData.contents : "{}";
    const body = JSON.parse(raw);
    const isSet = body.type === "set";
    const key = isSet ? ("SET:" + body.name) : (body.grade + "_" + body.term);
    const data = sheet.getDataRange().getValues();
    let rowIndex = -1;
    
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]) === key) {
        rowIndex = i + 1;
        break;
      }
    }
    
    const rowValues = isSet
      ? [key, body.name, JSON.stringify(body.data), new Date()]
      : [key, body.title, JSON.stringify(body.questions), new Date()];
    if (rowIndex > 0) {
      sheet.getRange(rowIndex, 1, 1, 4).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }
    
    return jsonResponse({ status: "saved", key: key });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("漢字プリントDB");
  if (!sheet) {
    sheet = ss.insertSheet("漢字プリントDB");
    sheet.appendRow(["学年_学期", "プリント題名", "問題JSON", "更新日時"]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

【初期設定の手順（初回のみ）】

  1. Googleスプレッドシートを開き、上部メニュー「拡張機能」＞「Apps Script」をクリック
  2. 元のコードを消して上のコードを貼り付け、フロッピーアイコン（保存）を押す
  3. 右上の青い「デプロイ」＞「新しいデプロイ」をクリック
  4. 種類の歯車アイコンから「ウェブアプリ」を選択し、アクセスできるユーザーを「全員」にしてデプロイ
  5. 発行された「ウェブアプリのURL」をコピーして上の入力欄に貼り付けて保存
